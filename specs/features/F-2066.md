### 5.1 타입

```ts
import type { Dispatch, RefObject, SetStateAction } from 'react'

export type UseSidebarLayoutOptions = {
  sidebarRef: RefObject<HTMLElement | null>
  appShellRef: RefObject<HTMLDivElement | null>
  toggleButtonRef: RefObject<HTMLButtonElement | null>
  mapRoute: object | null
  settingsOpen: boolean
  searchOpen: boolean
  paletteOpen: boolean
  deleteTarget: object | null
  moveDocTarget: object | null
  bulkDeleteItems: object | null
}

export type UseSidebarLayoutResult = {
  narrow: boolean
  sidebarOpen: boolean
  setSidebarOpen: Dispatch<SetStateAction<boolean>>
  sidebarCollapsed: boolean
  openFolders: string[]
  addOpenFolders: (ids: string[] | null | undefined) => void
  toggleFolderOpen: (id: string) => void
  collapseAllFolders: () => void
  onNavigateFolder: (folderId: string) => void
  closeSidebarIfNarrow: () => void
  toggleSidebar: () => void
```

- **옵션·반환 이름은 지금 App 변수 이름과 같다.** 호출이 `{ sidebarRef, appShellRef, … }` 단축 표기 한 벌, 반환이 `const { narrow, … } = useSidebarLayout(…)` 한 벌로 끝나고, 2.3 의 호출처가 한 글자도 안 바뀐다. 이름을 바꾸지 않는다

## 11. 열린 질문 (모두 기본값, 사람 결정 필요 없음)

### Q1. ref 세 개(`narrowRef`·`sidebarOpenRef`·`sidebarCollapsedRef`)를 훅으로 옮기나 — **기본값: 옮긴다 (4.2)**

- 이유: 선언·대입·읽기가 모두 이 구역이라 F-2059 3.2 의 옮기는 조건에 맞는다. 읽는 곳이 클릭 핸들러뿐이라 대입 effect 가 앞당겨져도 볼 수 있는 차이가 없다. App 의 최신값 effect ③ 이 이 구역을 몰라도 되게 된다
- 다른 쪽(App 에 두고 ref 객체 3개를 인자로): ③ 이 그대로이고 새 effect 가 안 생기지만, 옵션이 3개 늘고 `onNavigateFolder` deps 가 `[scrollToFolderRow, narrowRef, sidebarOpenRef, sidebarCollapsedRef]` 로 바뀐다(`exhaustive-deps`). App 이 이 구역 상태를 계속 ref 로 들고 있게 된다
