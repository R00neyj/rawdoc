### 2.3 모듈 계약

```ts
// src/app/appNotices.ts
import type { Notice } from './notice'
import type { E2eeConvertDirection, E2eeConvertOutcome, E2eeConvertStopReason } from '../e2ee/convert'
import { isE2eeStoreError } from '../e2ee/e2eeStore'
import { ACCOUNT_BLOCKED_MESSAGE, formatCount, formatResetTime } from '../lib/usageLimits'

export const READ_ONLY_LIVE_NOTICE // as const 그대로 — revoked·signedOut·disconnected
export const LIVE_NOTICE          // as const 그대로 — forbidden·revoked·gone·signedOut·disconnected·disconnectedVolatile·offlineView·merged·tooLarge
export const E2EE_NOTICE          // as const 그대로 — folderRule·locked·unavailable·tooManyRefs·createTooLarge·saveTooLarge
export const E2EE_CONVERT_NOTICE  // as const 그대로 — offline·busy·insideFolder
export const E2EE_CONVERT_STOP_REASON: Record<Exclude<E2eeConvertStopReason, 'cancelled' | 'day-limit' | 'account-blocked'>, string>
export function e2eeConvertProgressText(direction: E2eeConvertDirection, done: number, total: number): string
export function e2eeConvertResultNotice(outcome: E2eeConvertOutcome, direction: E2eeConvertDirection, targetKind: 'doc' | 'folder', name: string): Notice
export function e2eeCreateErrorMessage(err: unknown): string | null

// src/app/docMeta.ts
import type { Doc, LineEnding } from '../types'

export type DocMeta = Pick<Doc, 'id' | 'title' | 'updatedAt' | 'folderId' | 'pinnedAt' | 'role' | 'ownerEmail' | 'viaFolder' | 'e2ee'> & { lineEnding?: LineEnding }
export type OpenDoc = { id: string; content: string; lineEnding: LineEnding }
export function stripContent(doc: Doc): DocMeta
export function isSharedDoc(doc: Pick<DocMeta, 'role'> | null | undefined): boolean
export function sortByUpdatedAtDesc<T extends { updatedAt: number }>(list: T[]): T[]
```

- 위 블록은 **export 이름과 시그니처의 계약**이다. 상수의 값·함수 본문은 `App.tsx` 에서 바이트 그대로 가져온다(3.6). `as const` 를 명시 타입으로 바꾸지 않는다
- 두 파일은 React·DOM·`window`·`location` 을 쓰지 않는다. 런타임 import 는 `appNotices.ts` 의 `e2eeStore`·`usageLimits` 두 개뿐이고, 둘 다 `App.tsx` 가 이미 import 하므로 번들 청크 구성은 바뀌지 않는다
- **두 파일은 `App.tsx` 를 import 하지 않는다** (순환 금지)
- 머리 주석은 파일마다 한 줄만: `// App 알림 띠 문구·금고 옮기기 결과 알림 (F-2059 2.3, F-2060)`, `// 사이드바 목록 메타 타입과 정리 함수 (F-2059 2.3, F-2060)`

## 4. `Stats`·`AppNotice` — 개요 목록에 없던 두 타입
